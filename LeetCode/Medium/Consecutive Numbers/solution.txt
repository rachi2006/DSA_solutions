import pandas as pd
def consecutive_numbers(logs: pd.DataFrame) -> pd.DataFrame:
    logs = logs.sort_values('id')
    next_num = logs['num'].shift(-1)
    next_next_num = logs['num'].shift(-2)
    condition = (logs['num'] == next_num) & (logs['num'] == next_next_num)
    result = logs[condition]['num'].unique()
    return pd.DataFrame({'ConsecutiveNums': result})
